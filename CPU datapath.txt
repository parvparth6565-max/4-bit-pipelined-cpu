// ============================================================================
// Module: instruction_memory
// ============================================================================
module instruction_memory(
  input [7:0] pc_IF,
  output [18:0] instruction_IF
);
  parameter DEPTH = 226;

  reg [18:0] mem[0:DEPTH+1];
  integer i;

  initial begin
    // Initialize all memory locations to HALT by default
    for (i = 0; i < DEPTH+2; i = i + 1)
      mem[i] = 19'b1111111111111111111;

    // Example Test Program:
    // Format: {imm[4], sel[3], rA[2], rB[2], wSel[2], mw, mr, m2r, we, alu_src, br}
    
    // mem[0]: BEQ R1, R2, +2 
    // Compares R1 and R2 using SUB (sel = 3'b001). If equal, branch +2.
    mem[0] = 19'b0010_001_01_10_00_0_0_0_0_0_1; 

    // mem[1]: ADD R0, R1, R2 (Executed if branch not taken)
    mem[1] = 19'b0000_000_01_10_00_0_0_0_1_0_0; 

    // mem[2]: ADD R0, R0, R1
    mem[2] = 19'b0000_000_00_01_00_0_0_0_1_0_0; 

    // mem[3]: HALT
    mem[3] = 19'b1111111111111111111; 
  end

  assign instruction_IF = mem[pc_IF];
endmodule
