module data_memory(
  input clk,
  input mem_write,
  input mem_read,
  input [3:0] address,
  input [3:0] write_data,
  output reg [3:0] read_data
);
  reg [3:0] mem[0:15];

  initial begin
    mem[0] = 4'd2;
    mem[1] = 4'd4;
    mem[2] = 4'd6;
    mem[3] = 4'd8;
    mem[4] = 4'd10;
    mem[5] = 4'd12;
    mem[6] = 4'd14;
    mem[7] = 4'd1;
  end

  always @(posedge clk) begin
    if (mem_write)
      mem[address] <= write_data;
  end

  always @(*) begin
    if (mem_read)
      read_data = mem[address];
    else
      read_data = 4'b0000;
  end
endmodule
