module forwarding_unit(
  input [1:0] select_A_EX,
  input [1:0] select_B_EX,
  input [1:0] write_select_MEM,
  input [1:0] write_select_WB,
  input write_enable_MEM,
  input write_enable_WB,
  output reg [1:0] forward_A,
  output reg [1:0] forward_B
);
  always @(*) begin
    forward_A = 2'b00;
    forward_B = 2'b00;

    if (write_enable_MEM && (write_select_MEM == select_A_EX))
      forward_A = 2'b10;

    if (write_enable_MEM && (write_select_MEM == select_B_EX))
      forward_B = 2'b10;

    if (write_enable_WB && !(write_enable_MEM && (write_select_MEM == select_A_EX)) && (write_select_WB == select_A_EX))
      forward_A = 2'b01;

    if (write_enable_WB && !(write_enable_MEM && (write_select_MEM == select_B_EX)) && (write_select_WB == select_B_EX))
      forward_B = 2'b01;
  end
endmodule
