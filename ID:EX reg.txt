module ID_EX_reg(
  input clk, rst, flush,
  input [1:0] select_A_in,
  input [1:0] select_B_in,
  input [2:0] sel_in,
  input [3:0] immediate_in,
  input mem_write_in,
  input mem_read_in,
  input mem_to_reg_in,
  input write_enable_in,
  input alu_src_in,
  input branch_in,
  input halt_in,
  input [1:0] write_select_in,
  input [3:0] write_data_in,
  input [3:0] data_A_in,
  input [3:0] data_B_in,

  output reg [1:0] select_A_out,
  output reg [1:0] select_B_out,
  output reg [2:0] sel_out,
  output reg [1:0] write_select_out,
  output reg [3:0] immediate_out,
  output reg mem_write_out,
  output reg mem_read_out,
  output reg mem_to_reg_out,
  output reg write_enable_out,
  output reg alu_src_out,
  output reg branch_out,
  output reg halt_out,
  output reg [3:0] write_data_out,
  output reg [3:0] data_A_out,
  output reg [3:0] data_B_out
);
  always @(posedge clk or posedge rst) begin
    if (rst || flush) begin
      select_A_out     <= 2'b00;
      select_B_out     <= 2'b00;
      sel_out          <= 3'b000;
      write_select_out <= 2'b00;
      immediate_out    <= 4'b0000;
      mem_write_out    <= 1'b0;
      mem_read_out     <= 1'b0;
      mem_to_reg_out   <= 1'b0;
      write_enable_out <= 1'b0;
      alu_src_out      <= 1'b0;
      branch_out       <= 1'b0;
      halt_out         <= 1'b0;
      write_data_out   <= 4'b0000;
      data_A_out       <= 4'b0000;
      data_B_out       <= 4'b0000;
    end else begin
      select_A_out     <= select_A_in;
      select_B_out     <= select_B_in;
      sel_out          <= sel_in;
      write_select_out <= write_select_in;
      immediate_out    <= immediate_in;
      mem_write_out    <= mem_write_in;
      mem_read_out     <= mem_read_in;
      mem_to_reg_out   <= mem_to_reg_in;
      write_enable_out <= write_enable_in;
      alu_src_out      <= alu_src_in;
      branch_out       <= branch_in;
      halt_out         <= halt_in;
      write_data_out   <= write_data_in;
      data_A_out       <= data_A_in;
      data_B_out       <= data_B_in;
    end
  end
endmodule
